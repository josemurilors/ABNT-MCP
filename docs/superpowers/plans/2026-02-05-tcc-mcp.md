**Goal:** Build a Python MCP server with 7 academic writing tools for Brazilian TCCs following ABNT standards.

**Architecture:** FastMCP server exposing 7 tools (search_papers, get_bibtex_by_doi, clean_bibtex, read_pdf_paper, validate_citations, check_tcc_structure, compile_document). OpenCode skill provides high-level commands (#pesquisar, #citar, #escrever, etc.) that map to MCP tools. Cross-platform via uv run.

**Tech Stack:** Python 3.8+, FastMCP, bibtexparser, requests, pypdf, uv (package manager)

## Global Constraints

- Python 3.8+ compatibility
- Cross-platform: Windows, Linux, macOS (no bash scripts for setup)
- All file readers must use UTF-8 with fallback to latin-1/cp1252
- API requests must include User-Agent with configurable email via TCC_MCP_EMAIL env var
- Preflight checks before compilation (verify binaries exist)
- Recursive file scanning for LaTeX projects with \input{} and \include{}
- Anti-duplication when inserting BibTeX entries

```python
from pathlib import Path

def read_file_with_fallback(file_path: str) -> str:
    """Read file with UTF-8 encoding, fallback to latin-1 and cp1252."""
    path = Path(file_path)
    if not path.exists():
        raise FileNotFoundError(f"File not found: {file_path}")
    
    for encoding in ["utf-8", "latin-1", "cp1252"]:
        try:
            with open(file_path, "r", encoding=encoding) as f:
                return f.read()
        except UnicodeDecodeError:
            continue
    
    raise ValueError(f"Could not decode file {file_path} with any supported encoding")

def write_file_with_encoding(file_path: str, content: str, encoding: str = "utf-8"):
    """Write file with specified encoding."""
    with open(file_path, "w", encoding=encoding) as f:
        f.write(content)
```

```python
import os
import requests
from typing import Optional

def get_user_agent() -> str:
    """Get User-Agent with configured email."""
    email = os.getenv("TCC_MCP_EMAIL", "usuario-tcc@academico.local")
    return f"TCC-MCP/1.0 (https://github.com/usuario/tcc-mcp; mailto:{email})"

def search_papers(query: str, limit: int = 10, source: str = "crossref") -> dict:
    """Search academic papers via Crossref or Semantic Scholar."""
    headers = {
        "User-Agent": get_user_agent(),
        "Accept": "application/json"
    }
    
    try:
        if source == "crossref":
            return _search_crossref(query, limit, headers)
        elif source == "semantic_scholar":
            return _search_semantic_scholar(query, limit, headers)
        else:
            return {"status": "error", "message": f"Unknown source: {source}"}
    except Exception as e:
        return {"status": "error", "message": str(e)}

def _search_crossref(query: str, limit: int, headers: dict) -> dict:
    """Search via Crossref API."""
    url = "https://api.crossref.org/works"
    params = {
        "query": query,
        "rows": limit,
        "select": "DOI,title,author,issued,abstract"
    }
    
    response = requests.get(url, params=params, headers=headers, timeout=10)
    response.raise_for_status()
    
    data = response.json()
    papers = []
    
    for item in data.get("message", {}).get("items", []):
        paper = {
            "doi": item.get("DOI", ""),
            "title": item.get("title", [""])[0] if item.get("title") else "",
            "authors": _format_authors(item.get("author", [])),
            "year": _extract_year(item.get("issued", {})),
            "abstract": item.get("abstract", "")
        }
        papers.append(paper)
    
    return {"status": "success", "papers": papers}

def _search_semantic_scholar(query: str, limit: int, headers: dict) -> dict:
    """Search via Semantic Scholar API."""
    url = "https://api.semanticscholar.org/graph/v1/paper/search"
    params = {
        "query": query,
        "limit": limit,
        "fields": "title,authors,year,abstract,externalIds"
    }
    
    response = requests.get(url, params=params, headers=headers, timeout=10)
    response.raise_for_status()
    
    data = response.json()
    papers = []
    
    for item in data.get("data", []):
        paper = {
            "doi": item.get("externalIds", {}).get("DOI", ""),
            "title": item.get("title", ""),
            "authors": ", ".join([a.get("name", "") for a in item.get("authors", [])]),
            "year": item.get("year", 0),
            "abstract": item.get("abstract", "")
        }
        papers.append(paper)
    
    return {"status": "success", "papers": papers}

def _format_authors(authors: list) -> str:
    """Format author list to string."""
    names = []
    for author in authors:
        given = author.get("given", "")
        family = author.get("family", "")
        names.append(f"{given} {family}".strip())
    return ", ".join(names)

def _extract_year(issued: dict) -> int:
    """Extract year from issued date."""
    date_parts = issued.get("date-parts", [[]])
    if date_parts and date_parts[0]:
        return date_parts[0][0]
    return 0
```

```python
import os
import re
import requests
from pathlib import Path
from typing import Optional
import bibtexparser
from ..utils import read_file_with_fallback, write_file_with_encoding

def get_user_agent() -> str:
    email = os.getenv("TCC_MCP_EMAIL", "usuario-tcc@academico.local")
    return f"TCC-MCP/1.0 (https://github.com/usuario/tcc-mcp; mailto:{email})"

def get_bibtex_by_doi(doi: str, bib_path: Optional[str] = None) -> dict:
    """Fetch BibTeX by DOI and optionally insert into .bib file."""
    headers = {
        "User-Agent": get_user_agent(),
        "Accept": "application/x-bibtex"
    }
    
    try:
        url = f"https://doi.org/{doi}"
        response = requests.get(url, headers=headers, timeout=10)
        response.raise_for_status()
        
        bibtex = response.text
        key = extract_bibtex_key(bibtex)
        
        if bib_path:
            existing_keys = parse_bib_keys(bib_path)
            if key in existing_keys:
                return {"status": "duplicate", "message": f"Key '{key}' already exists", "key": key}
            
            append_to_bib(bib_path, bibtex)
            return {"status": "added", "key": key, "bibtex": bibtex}
        
        return {"status": "fetched", "key": key, "bibtex": bibtex}
    
    except Exception as e:
        return {"status": "error", "message": str(e)}

def extract_bibtex_key(bibtex: str) -> str:
    """Extract the citation key from BibTeX entry."""
    match = re.search(r'@\w+\{([^,]+),', bibtex)
    return match.group(1).strip() if match else ""

def parse_bib_keys(bib_path: str) -> set:
    """Parse all citation keys from .bib file."""
    content = read_file_with_fallback(bib_path)
    keys = set()
    
    for match in re.finditer(r'@\w+\{([^,]+),', content):
        keys.add(match.group(1).strip())
    
    return keys

def append_to_bib(bib_path: str, bibtex: str):
    """Append BibTeX entry to file."""
    content = read_file_with_fallback(bib_path)
    if content and not content.endswith('\n'):
        content += '\n'
    content += bibtex + '\n'
    write_file_with_encoding(bib_path, content)

def clean_bibtex(bib_path: str) -> dict:
    """Clean and standardize BibTeX file for ABNT."""
    try:
        content = read_file_with_fallback(bib_path)
        bib_database = bibtexparser.loads(content)
        
        cleaned_entries = []
        missing_fields_report = []
        
        for entry in bib_database.entries:
            entry_id = entry.get('ID', '')
            
            entry['ID'] = entry_id.lower().replace(' ', '_')
            
            if 'author' in entry:
                entry['author'] = clean_author_names(entry['author'])
            
            if 'title' in entry:
                entry['title'] = escape_tex_accents(entry['title'])
            
            missing = check_abnt_required_fields(entry)
            if missing:
                missing_fields_report.append({
                    "key": entry['ID'],
                    "missing_fields": missing
                })
            
            cleaned_entries.append(entry)
        
        cleaned_db = bibtexparser.bibdatabase.BibDatabase()
        cleaned_db.entries = cleaned_entries
        cleaned_bibtex = bibtexparser.dumps(cleaned_db)
        
        write_file_with_encoding(bib_path, cleaned_bibtex)
        
        return {
            "status": "cleaned",
            "cleaned_bibtex": cleaned_bibtex,
            "missing_fields_report": missing_fields_report
        }
    
    except Exception as e:
        return {"status": "error", "message": str(e)}

def clean_author_names(author_str: str) -> str:
    """Convert UPPERCASE names to Title Case."""
    if author_str.isupper():
        return author_str.title()
    return author_str

def escape_tex_accents(text: str) -> str:
    """Escape accents for TeX."""
    replacements = {
        'ã': r'{\~a}', 'õ': r'{\~o}',
        'á': r'{\'a}', 'é': r'{\'e}', 'í': r'{\'i}', 'ó': r'{\'o}', 'ú': r'{\'u}',
        'â': r'{\^a}', 'ê': r'{\^e}', 'ô': r'{\^o}',
        'ç': r'{\c c}'
    }
    for char, tex in replacements.items():
        text = text.replace(char, tex)
    return text

def check_abnt_required_fields(entry: dict) -> list:
    """Check for ABNT required fields."""
    entry_type = entry.get('ENTRYTYPE', '').lower()
    
    required = {
        'article': ['author', 'title', 'journal', 'year', 'volume', 'number', 'pages'],
        'book': ['author', 'title', 'year', 'publisher', 'address'],
        'inproceedings': ['author', 'title', 'booktitle', 'year'],
        'phdthesis': ['author', 'title', 'year', 'school'],
        'mastersthesis': ['author', 'title', 'year', 'school']
    }
    
    required_fields = required.get(entry_type, ['author', 'title', 'year'])
    missing = [f for f in required_fields if f not in entry or not entry[f]]
    
    return missing
```

```python
from pathlib import Path
from typing import Optional
from pypdf import PdfReader

def read_pdf_paper(file_path: str, pages: Optional[list[int]] = None) -> dict:
    """Extract text from PDF file."""
    path = Path(file_path)
    
    if not path.exists():
        return {"status": "error", "message": f"File not found: {file_path}"}
    
    if not path.suffix.lower() == '.pdf':
        return {"status": "error", "message": "File is not a PDF"}
    
    try:
        reader = PdfReader(file_path)
        total_pages = len(reader.pages)
```

```python
import re
from pathlib import Path
from typing import Optional
from ..utils import read_file_with_fallback

def validate_citations(project_path: str) -> dict:
    """Validate citations in LaTeX project (recursive)."""
    try:
        project = Path(project_path)
        
        main_tex = find_main_tex(project)
        if not main_tex:
            return {"status": "error", "message": "No .tex file found"}
        
        tex_files = resolve_tex_files(main_tex)
        
        all_citations = set()
        for tex_file in tex_files:
            content = read_file_with_fallback(str(tex_file))
            citations = extract_citations(content)
            all_citations.update(citations)
        
        bib_file = find_bib_file(project)
        if not bib_file:
            return {"status": "error", "message": "No .bib file found"}
        
        bib_keys = parse_bib_keys(bib_file)
        
        orphan_citations = list(all_citations - bib_keys)
        uncited_references = list(bib_keys - all_citations)
        
        return {
            "status": "success",
            "orphan_citations": orphan_citations,
            "uncited_references": uncited_references,
            "total_citations": len(all_citations),
            "total_references": len(bib_keys)
        }
    
    except Exception as e:
        return {"status": "error", "message": str(e)}

def find_main_tex(project: Path) -> Optional[Path]:
    """Find main .tex file in project."""
    for name in ["main.tex", "tcc.tex", "dissertacao.tex", "tese.tex"]:
        main = project / name
        if main.exists():
            return main
    
    tex_files = list(project.glob("*.tex"))
    return tex_files[0] if tex_files else None

def find_bib_file(project: Path) -> Optional[Path]:
    """Find .bib file in project."""
    for name in ["referencias.bib", "references.bib", "bibliografia.bib"]:
        bib = project / name
        if bib.exists():
            return bib
    
    bib_files = list(project.glob("*.bib"))
    return bib_files[0] if bib_files else None

def resolve_tex_files(main_tex: Path) -> list[Path]:
    """Recursively resolve all .tex files via \input{} and \include{}."""
    files = [main_tex]
    content = read_file_with_fallback(str(main_tex))
    
    for match in re.finditer(r'\\(?:input|include)\{([^}]+)\}', content):
        included = match.group(1)
        if not included.endswith('.tex'):
            included += '.tex'
        
        full_path = main_tex.parent / included
        if full_path.exists():
            files.extend(resolve_tex_files(full_path))
    
    return files

def extract_citations(content: str) -> set:
    """Extract all citation keys from LaTeX content."""
    citations = set()
    
    patterns = [
        r'\\cite\{([^}]+)\}',
        r'\\citeauthor\{([^}]+)\}',
        r'\\citeyear\{([^}]+)\}',
        r'\\citet\{([^}]+)\}',
        r'\\citep\{([^}]+)\}'
    ]
    
    for pattern in patterns:
        for match in re.finditer(pattern, content):
            keys = match.group(1).split(',')
            citations.update(key.strip() for key in keys)
    
    return citations

def parse_bib_keys(bib_path: Path) -> set:
    """Parse all citation keys from .bib file."""
    content = read_file_with_fallback(str(bib_path))
    keys = set()
    
    for match in re.finditer(r'@\w+\{([^,]+),', content):
        keys.add(match.group(1).strip())
    
    return keys
```

```python
import re
from pathlib import Path
from ..utils import read_file_with_fallback

REQUIRED_ELEMENTS = {
    "pre-textual": [
        "capa", "folha_de_rosto", "resumo", "abstract",
        "lista_ilustracoes", "lista_tabelas", "lista_abreviaturas", "sumario"
    ],
    "textual": [
        "introducao", "desenvolvimento", "conclusao"
    ],
    "post-textual": [
        "referencias"
    ],
    "optional": [
        "apendices", "anexos"
    ]
}

ELEMENT_PATTERNS = {
    "capa": [r"\\begin\{titlepage\}", r"\\maketitle", r"capa", r"cover"],
    "folha_de_rosto": [r"folha\s+de\s+rosto", r"title\s+page"],
    "resumo": [r"\\begin\{resumo\}", r"\\chapter\*?\{resumo\}", r"resumo"],
    "abstract": [r"\\begin\{abstract\}", r"\\chapter\*?\{abstract\}", r"abstract"],
    "lista_ilustracoes": [r"\\listoffigures", r"lista\s+de\s+ilustrações", r"lista\s+de\s+figuras"],
    "lista_tabelas": [r"\\listoftables", r"lista\s+de\s+tabelas"],
    "lista_abreviaturas": [r"lista\s+de\s+abreviaturas", r"lista\s+de\s+siglas"],
    "sumario": [r"\\tableofcontents", r"sumário", r"sumario"],
    "introducao": [r"\\chapter\*?\{introdução\}", r"\\chapter\*?\{introducao\}", r"introdução"],
    "desenvolvimento": [r"\\chapter", r"fundamentação", r"metodologia", r"resultados"],
    "conclusao": [r"\\chapter\*?\{conclusão\}", r"\\chapter\*?\{considerações\s+finais\}", r"conclusão"],
    "referencias": [r"\\bibliography", r"\\begin\{thebibliography\}", r"referências", r"referencias"],
    "apendices": [r"\\begin\{apendice\}", r"\\chapter\*?\{apêndice\}", r"apêndices"],
    "anexos": [r"\\begin\{anexo\}", r"\\chapter\*?\{anexo\}", r"anexos"]
}

def check_tcc_structure(project_path: str) -> dict:
    """Check TCC structure according to NBR 14724."""
    try:
        project = Path(project_path)
        
        all_content = collect_all_content(project)
        
        found_elements = set()
        for element, patterns in ELEMENT_PATTERNS.items():
            for pattern in patterns:
                if re.search(pattern, all_content, re.IGNORECASE):
                    found_elements.add(element)
                    break
        
        missing_elements = []
        for category, elements in REQUIRED_ELEMENTS.items():
            if category == "optional":
                continue
            for element in elements:
                if element not in found_elements:
                    missing_elements.append(element)
        
        return {
            "status": "success",
            "found_elements": sorted(list(found_elements)),
            "missing_elements": missing_elements,
            "has_optional_apendices": "apendices" in found_elements,
            "has_optional_anexos": "anexos" in found_elements
        }
    
    except Exception as e:
        return {"status": "error", "message": str(e)}

def collect_all_content(project: Path) -> str:
    """Collect content from all .tex files in project."""
    all_content = []
    
    for tex_file in project.glob("**/*.tex"):
        try:
            content = read_file_with_fallback(str(tex_file))
            all_content.append(content)
        except:
            continue
    
    return "\n".join(all_content)
```

```python
import shutil
import subprocess
from pathlib import Path
from ..utils import read_file_with_fallback

def compile_document(project_path: str, engine: str = "auto") -> dict:
    """Compile document with preflight check."""
    try:
        project = Path(project_path)
        
        if engine == "auto":
            engine = detect_engine(project)
            if not engine:
                return {
                    "status": "error",
                    "message": "No compiler found. Install TeX Live, Pandoc, or Typst."
                }
        
        availability = check_compiler_availability(engine)
        if not availability["available"]:
            return availability
        
        if engine in ["latexmk", "pdflatex"]:
            return compile_latex(project, engine)
        elif engine == "pandoc":
            return compile_pandoc(project)
        elif engine == "typst":
            return compile_typst(project)
        else:
            return {"status": "error", "message": f"Unknown engine: {engine}"}
    
    except Exception as e:
        return {"status": "error", "message": str(e)}

def detect_engine(project: Path) -> str:
    """Detect compilation engine from project files."""
    if (project / "main.tex").exists() or any(project.glob("*.tex")):
        if shutil.which("latexmk"):
            return "latexmk"
        elif shutil.which("pdflatex"):
            return "pdflatex"
    
    if (project / "main.md").exists() and shutil.which("pandoc"):
        return "pandoc"
    
    if (project / "main.typ").exists() and shutil.which("typst"):
        return "typst"
    
    return None

def check_compiler_availability(engine: str) -> dict:
    """Check if compiler binary is available."""
    binary_map = {
        "latexmk": "latexmk",
        "pdflatex": "pdflatex",
        "pandoc": "pandoc",
        "typst": "typst"
    }
    
    binary = binary_map.get(engine)
    if not binary:
        return {"available": False, "message": f"Unknown engine: {engine}"}
    
    if not shutil.which(binary):
        return {
            "available": False,
            "message": f"Compiler '{binary}' not found in PATH. Please install it."
        }
    
    return {"available": True, "engine": binary}

def compile_latex(project: Path, engine: str) -> dict:
    """Compile LaTeX document."""
    main_tex = find_main_tex(project)
    if not main_tex:
        return {"status": "error", "message": "No .tex file found"}
    
    try:
        result = subprocess.run(
            [engine, "-interaction=nonstopmode", str(main_tex)],
            cwd=str(project),
            capture_output=True,
            text=True,
            timeout=60
        )
        
        if result.returncode == 0:
            pdf_path = main_tex.with_suffix('.pdf')
            return {
                "status": "success",
                "pdf_path": str(pdf_path),
                "message": "Compilation successful"
            }
        else:
            errors = extract_latex_errors(result.stdout)
            return {
                "status": "error",
                "message": "Compilation failed",
                "errors": errors
            }
    
    except subprocess.TimeoutExpired:
        return {"status": "error", "message": "Compilation timeout (60s)"}

def compile_pandoc(project: Path) -> dict:
    """Compile Markdown with Pandoc."""
    main_md = project / "main.md"
    if not main_md.exists():
        return {"status": "error", "message": "main.md not found"}
    
    output_pdf = project / "output.pdf"
    
    try:
        result = subprocess.run(
            ["pandoc", str(main_md), "-o", str(output_pdf)],
            cwd=str(project),
            capture_output=True,
            text=True,
            timeout=60
        )
        
        if result.returncode == 0:
            return {
                "status": "success",
                "pdf_path": str(output_pdf),
                "message": "Compilation successful"
            }
        else:
            return {
                "status": "error",
                "message": "Pandoc compilation failed",
                "errors": result.stderr
            }
    
    except subprocess.TimeoutExpired:
        return {"status": "error", "message": "Compilation timeout (60s)"}

def compile_typst(project: Path) -> dict:
    """Compile Typst document."""
    main_typ = project / "main.typ"
    if not main_typ.exists():
        return {"status": "error", "message": "main.typ not found"}
    
    output_pdf = project / "output.pdf"
    
    try:
        result = subprocess.run(
            ["typst", "compile", str(main_typ), str(output_pdf)],
            cwd=str(project),
            capture_output=True,
            text=True,
            timeout=60
        )
        
        if result.returncode == 0:
            return {
                "status": "success",
                "pdf_path": str(output_pdf),
                "message": "Compilation successful"
            }
        else:
            return {
                "status": "error",
                "message": "Typst compilation failed",
                "errors": result.stderr
            }
    
    except subprocess.TimeoutExpired:
        return {"status": "error", "message": "Compilation timeout (60s)"}

def find_main_tex(project: Path) -> Path:
    """Find main .tex file."""
    for name in ["main.tex", "tcc.tex", "dissertacao.tex"]:
        main = project / name
        if main.exists():
            return main
    
    tex_files = list(project.glob("*.tex"))
    return tex_files[0] if tex_files else None

def extract_latex_errors(log: str) -> list:
    """Extract error lines from LaTeX log."""
    errors = []
    for line in log.split('\n'):
        if line.startswith('!') or 'Error' in line:
            errors.append(line)
    return errors[:10]
```

```markdown
## ABNT Standards

### NBR 10520:2023 (Citations)
- **Direct citation (short):** "Text" (AUTHOR, Year, p. X)
- **Direct citation (long):** Indented block, size 10, single spacing
- **Indirect citation:** AUTHOR (Year) or (AUTHOR, Year)

### NBR 6023 (References)
- Author surnames in UPPERCASE
- Title in bold
- Required fields vary by type (article, book, etc.)

### NBR 14724 (Structure)
**Pre-textual:** Capa, Folha de Rosto, Resumo, Abstract, Lista de Ilustrações, Lista de Tabelas, Lista de Abreviaturas, Sumário
**Textual:** Introdução, Desenvolvimento, Conclusão
**Post-textual:** Referências, Apêndices (optional), Anexos (optional)
```
